cmake_minimum_required(VERSION 3.16)
project(sem7 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(THREADS_PREFER_PTHREAD_FLAG TRUE)
find_package(Threads REQUIRED)

add_executable(sem7 main.cpp)
target_link_libraries(sem7 PRIVATE Threads::Threads)
if(MSVC)
    target_compile_options(sem7 PRIVATE /W4 /WX)
else()
    target_compile_options(sem7 PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()

include(CTest)
if(BUILD_TESTING)
    add_test(NAME serial COMMAND sem7 1)
    set_tests_properties(serial PROPERTIES PASS_REGULAR_EXPRESSION "serial.*check OK")
    add_test(NAME gated COMMAND sem7 3)
    set_tests_properties(gated PROPERTIES PASS_REGULAR_EXPRESSION "gated.*check OK")
    add_test(NAME invalid_stage COMMAND sem7 7)
    add_test(NAME invalid_baseline COMMAND sem7 6 0)
    add_test(NAME unexpected_argument COMMAND sem7 2 10)
    set_tests_properties(invalid_stage invalid_baseline unexpected_argument
        PROPERTIES WILL_FAIL TRUE)
    set_tests_properties(serial gated PROPERTIES TIMEOUT 120)
endif()
